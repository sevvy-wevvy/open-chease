# plugin.txt for a compiled plugin. you ship main.c only, open-chease builds it.
# lines starting with # get skipped, everything else is key = value. unknown keys get skipped too
# full list of keys with notes is in template/plugin.txt

name = C example
description = best move arrow, a menu item and a box, in plain C
version = 1.0
author = sev
url = https://github.com/sevvy-wevvy/open-chease/tree/plugin-template/c

# what needs to be installed. if theres no C compiler the Plugins tab says so with a link
lang = c

# build runs once inside this folder, and again whenever main.c changes.
# what it prints goes to build.log here, the last lines show in the Plugins tab if it fails.
# cc is gcc or clang on linux and mac (on mac it comes with the Xcode command line tools).
# windows has no cc, so it gets its own line, gcc there is mingw (winlibs.com or msys2)
build = cc -O2 -o myplugin main.c
build.windows = gcc -O2 -o myplugin.exe main.c

# then this starts the built program. windows wants the .exe on the end
run = myplugin
run.windows = myplugin.exe
